
(function(){
  // Ближайший четверг 20:00 по Москве (UTC+3, без перехода на летнее время)
  function nextRun(now){
    var msk = new Date(now.getTime() + 3*3600e3);        // «московские» часы в UTC-полях
    var d = new Date(Date.UTC(msk.getUTCFullYear(), msk.getUTCMonth(), msk.getUTCDate(), 20, 0, 0));
    var add = (4 - msk.getUTCDay() + 7) % 7;
    d.setUTCDate(d.getUTCDate() + add);
    // если пробежка уже закончилась (после 21:05), переносим на следующую неделю
    if (add === 0 && msk.getTime() > d.getTime() + 60*60e3) d.setUTCDate(d.getUTCDate() + 7);
    return { mskStart: d, running: add === 0 && msk >= d && msk.getTime() <= d.getTime() + 60*60e3, diff: d.getTime() - msk.getTime() };
  }
  var months = ["января","февраля","марта","апреля","мая","июня","июля","августа","сентября","октября","ноября","декабря"];
  function plural(n, a, b, c){ n = Math.abs(n) % 100; var n1 = n % 10; if (n > 10 && n < 20) return c; if (n1 > 1 && n1 < 5) return b; if (n1 === 1) return a; return c; }
  function tick(){
    var r = nextRun(new Date());
    document.getElementById("nextDate").textContent = "чт, " + r.mskStart.getUTCDate() + " " + months[r.mskStart.getUTCMonth()] + ", 20:00";
    var el = document.getElementById("nextIn");
    if (r.running) { el.textContent = "бежим прямо сейчас"; return; }
    var m = Math.floor(r.diff / 60e3), days = Math.floor(m / 1440), h = Math.floor((m % 1440) / 60), mm = m % 60;
    el.textContent = "через " + (days ? days + " " + plural(days,"день","дня","дней") + " " : "") + h + " ч " + mm + " мин";
  }
  tick(); setInterval(tick, 30e3);

  // Копирование координат
  var cb = document.getElementById("copyCoords");
  if (cb) cb.addEventListener("click", function(){
    var txt = document.getElementById("coords").textContent;
    function done(){ cb.textContent = "Скопировано"; setTimeout(function(){ cb.textContent = "Скопировать"; }, 1800); }
    function fallback(){ var r = document.createRange(); r.selectNodeContents(document.getElementById("coords")); var s = getSelection(); s.removeAllRanges(); s.addRange(r); cb.textContent = "Выделено, нажми Ctrl+C"; }
    try { navigator.clipboard.writeText(txt).then(done, fallback); } catch(e){ fallback(); }
  });

  // Уведомление о данных
  var KEY = "chanka_notice_ok", notice = document.getElementById("notice");
  var seen = false; try { seen = localStorage.getItem(KEY) === "1"; } catch(e){}
  if (!seen) notice.hidden = false;
  document.getElementById("noticeOk").addEventListener("click", function(){
    notice.hidden = true; try { localStorage.setItem(KEY, "1"); } catch(e){}
  });

  // Луч налобника следует за курсором
  var hero = document.getElementById("hero");
  var reduce = window.matchMedia && matchMedia("(prefers-reduced-motion: reduce)").matches;
  if (!reduce) hero.addEventListener("pointermove", function(e){
    var b = hero.getBoundingClientRect();
    hero.style.setProperty("--x", (e.clientX - b.left) + "px");
    hero.style.setProperty("--y", (e.clientY - b.top) + "px");
  });

  // Топографические горизонтали на фоне
  var cv = document.getElementById("topo"), ctx = cv.getContext("2d");
  function draw(){
    var w = hero.clientWidth, h = hero.clientHeight, dpr = Math.min(window.devicePixelRatio || 1, 2);
    cv.width = w * dpr; cv.height = h * dpr; ctx.setTransform(dpr,0,0,dpr,0,0);
    ctx.clearRect(0,0,w,h); ctx.lineWidth = 1;
    var cx = w * 0.78, cy = h * 0.42;
    for (var i = 1; i <= 26; i++){
      var base = i * Math.max(w, h) / 34;
      ctx.beginPath();
      for (var a = 0; a <= 360; a += 3){
        var t = a * Math.PI / 180;
        var r = base * (1 + 0.16*Math.sin(3*t + i*0.35) + 0.08*Math.cos(5*t - i*0.2));
        var x = cx + r * Math.cos(t) * 1.25, y = cy + r * Math.sin(t);
        a === 0 ? ctx.moveTo(x,y) : ctx.lineTo(x,y);
      }
      ctx.closePath();
      ctx.strokeStyle = i % 5 === 0 ? "rgba(142,161,151,.30)" : "rgba(142,161,151,.13)";
      ctx.stroke();
    }
  }
  draw(); window.addEventListener("resize", draw);
})();
